  n = length(q)
  order = sortperm(q .* ohat; rev=true)
  S = Int[]; pS = 0.0; σS = 0.0; R = 1.0
  for i in order
      q[i] * ohat[i] > R || break
      push!(S, i); pS += q[i]; σS += 1 / ohat[i]
      R = (pS >= 1 - 1e-15 || σS >= 1) ? 0.0 : (1 - pS) / (1 - σS)
  end
  f = zeros(n)
  for i in S
      f[i] = max(q[i] - R / ohat[i], 0.0)
  end

function solve_conic(m::ScenarioModel; solver::Symbol = :clarabel)
    solver in (:clarabel, :scs) || throw(ArgumentError("solver must be :clarabel or :scs, got :$solver"))
    S, J = size(m.X)
    pos = findall(>(0.0), m.π)                       # scenarios with π_s = 0 are excluded
    model = JuMP.Model(solver === :clarabel ? Clarabel.Optimizer : SCS.Optimizer)
    JuMP.set_silent(model)
    @variable(model, a[1:J] >= 0)
    @variable(model, t[1:length(pos)])
    @constraint(model, sum(a) == 1)
    for (k, s) in enumerate(pos)
        @constraint(model, [t[k], 1.0, sum(m.X[s, j] * a[j] for j in 1:J)] in MOI.ExponentialCone())  # t_s ≤ log V_s(a)
    end
    @objective(model, Max, sum(m.π[s] * t[k] for (k, s) in enumerate(pos)))
    JuMP.optimize!(model)
    st = JuMP.termination_status(model)
    st in (MOI.OPTIMAL, MOI.ALMOST_OPTIMAL) || throw(SolverError(string(st), "conic solve failed (:$solver)"))
    a_val = max.(JuMP.value.(a), 0.0)
    return a_val ./ sum(a_val)
end

function polish_log_optimal(X, π, a0; act_tol = 1e-6, newton_max = 50, outer_max = 10, step_tol = 1e-15, add_tol = 1e-12, tol = CERT_TOL)
    pos = findall(>(0.0), π); Xp = X[pos, :]; πp = π[pos]
    a = [x < act_tol ? 0.0 : x for x in a0]; a ./= sum(a)
    for _ in 1:outer_max
        A = findall(>(0.0), a)
        for _ in 1:newton_max
            V = Xp * a; XA = Xp[:, A]
            g = XA' * (πp ./ V)
            H = -(XA' * (XA .* (πp ./ V .^ 2)))
            k = length(A)
            K = [H ones(k); ones(1, k) 0.0]
            d = (pinv(K) * vcat(-g, 0.0))[1:k]
            t = 1.0
            accepted = false
            while t ≥ 1e-12
                an = copy(a); an[A] .+= t .* d
                if all(>(0.0), an[A]) && all(>(0.0), Xp * an)
                    accepted = true; break
                end
                t /= 2
            end
            if !accepted
                ratios = [d[i] < 0 ? a[A[i]] / -d[i] : Inf for i in 1:k]
                a[A[argmin(ratios)]] = 0.0; a ./= sum(a)
                break
            end
            a[A] .+= t .* d
            maximum(abs, t .* d) < step_tol && break
        end
        V = Xp * a; r = Xp' * (πp ./ V)
        add = [j for j in eachindex(a) if a[j] == 0.0 && r[j] > 1 + add_tol]
        isempty(add) && kkt_residual(X, a, π) ≤ tol && break
        for j in add; a[j] = 1e-8; end
        a ./= sum(a)
    end
    return a, kkt_residual(X, a, π)
end

z = 1.49, 6.4 s; it also reproduced the drawdown law Pr(min W/w0 ≤ x) = x^(2/c−1) to within Monte Carlo error for

```julia
function simulate_ruin(m::ScenarioModel, a; rng, trials::Integer, rounds::Integer, w0 = 1.0, min_stake = 0.0, lot = 0.0,
                       fixed_cost = 0.0, fixed_stake = nothing, target = Inf)
    S, J = size(m.X)
    (trials ≥ 1 && rounds ≥ 1) || throw(ArgumentError("trials and rounds must be ≥ 1"))
    length(a) == J || throw(ArgumentError("length(a) = $(length(a)) != number of assets $J"))
    (w0 > 0 && min_stake ≥ 0 && lot ≥ 0 && fixed_cost ≥ 0 && target > w0) ||
        throw(ArgumentError("need w0 > 0, min_stake ≥ 0, lot ≥ 0, fixed_cost ≥ 0, target > w0"))
    (fixed_stake === nothing || fixed_stake > 0) || throw(ArgumentError("fixed_stake must be > 0"))
    bets = m.has_cash ? (2:J) : (1:J); cum = cumsum(m.π)
    ruined = 0; maxdd = zeros(trials); final = zeros(trials); minf = zeros(trials)
    for t in 1:trials
        W = w0; peak = w0; dd = 0.0; ruin = false; lo = w0
        for _ in 1:rounds
            s = min(searchsortedfirst(cum, rand(rng) * cum[end]), S)
            if fixed_stake === nothing
                stakes = [a[j] * W for j in bets]
                lot > 0 && (stakes = floor.(stakes ./ lot) .* lot)
            else
                stakes = zeros(length(bets)); stakes[1] = min(fixed_stake, W)
            end
            stakes[stakes .< min_stake] .= 0.0                     # bets below the minimum are skipped
            placed = count(>(0.0), stakes)
            W = W - sum(stakes) - fixed_cost * placed + sum(stakes[i] * m.X[s, bets[i]] for i in eachindex(bets))
            peak = max(peak, W); dd = max(dd, 1 - W / peak); lo = min(lo, W)
            if W ≤ 0 || (min_stake > 0 && W < min_stake); ruin = true; break; end
            W ≥ target && break
        end
        ruined += ruin; maxdd[t] = dd; final[t] = max(W, 0.0); minf[t] = max(lo, 0.0) / w0
    end
    p = ruined / trials
    return (p_ruin = p, se = sqrt(p * (1 - p) / trials), drawdown_q = quantile(maxdd, [0.5, 0.9, 0.99]),
            final_q = quantile(final, [0.1, 0.5, 0.9]), min_fraction = minf)
end
```

stops at ruin, at `target`, or after `rounds`. `drawdown_q` are quantiles of the maximum drawdown from the running peak;
`min_fraction` holds each trial's lowest wealth as a fraction of `w0` (for the drawdown law, E4).

```julia
function crra_weights(Xp, πp, a, k)
    V = Xp * a
    lw = log.(πp) .- k .* log.(V); lw .-= maximum(lw); Q = exp.(lw); Q ./= sum(Q)
    return Q, V
end

function polish_crra(X, π, a0, k; act_tol = 1e-6, newton_max = 50, outer_max = 10, step_tol = 1e-15, add_tol = 1e-12, tol = CERT_TOL)
    k ≥ 0 || throw(ArgumentError("k = 1/λ must be ≥ 0"))
    pos = findall(>(0.0), π); Xp = X[pos, :]; πp = π[pos]
    a = [x < act_tol ? 0.0 : x for x in a0]; a ./= sum(a)
    if any(<=(0.0), Xp * a)
        a = max.(collect(Float64, a0), 0.0); a ./= sum(a)
        any(<=(0.0), Xp * a) && (a = 0.999 .* a .+ 0.001 / length(a))
    end
    for _ in 1:outer_max
        for _ in 1:newton_max
            A = findall(>(0.0), a); kk = length(A)
            Q, V = crra_weights(Xp, πp, a, k); Y = Xp ./ V
            r = Y' * Q; YA = Y[:, A]
            g = r[A]
            H = -(1 + k) .* (YA' * (YA .* Q)) .+ k .* (r[A] * r[A]')
            K = [H ones(kk); ones(1, kk) 0.0]
            d = (pinv(K) * vcat(-g, 0.0))[1:kk]
            tmax, iblk = 1.0, 0
            for i in 1:kk
                if d[i] < 0 && a[A[i]] + d[i] ≤ 0.0
                    ti = a[A[i]] / -d[i]; ti < tmax && ((tmax, iblk) = (ti, i))
                end
            end
            t = tmax; drop = iblk > 0
            while t ≥ 1e-12
                an = copy(a); an[A] .+= t .* d; drop && (an[A[iblk]] = 0.0)
                all(≥(0.0), an) && all(>(0.0), Xp * an) && break
                t /= 2; drop = false
            end
            t < 1e-12 && break
            a[A] .+= t .* d; drop && (a[A[iblk]] = 0.0)
            a .= max.(a, 0.0); a ./= sum(a)
            !drop && maximum(abs, t .* d) < step_tol && break
        end
        Q, V = crra_weights(Xp, πp, a, k); r = (Xp ./ V)' * Q
        add = [j for j in eachindex(a) if a[j] == 0.0 && r[j] > 1 + add_tol]
        isempty(add) && kkt_residual(Xp, a, Q) ≤ tol && break
        for j in add; a[j] = 1e-8; end
        a ./= sum(a)
    end
    Q, _ = crra_weights(Xp, πp, a, k)
    return a, kkt_residual(Xp, a, Q)
end
```

the returned residual is the KKT residual under Q, so for `k = 0` it equals the old log-optimal residual up to rounding.)

  Validate `ε ≥ 0` and `Xt * a .> 0` (`ArgumentError`). Exact algorithm (prototyped 2026-11-03 on 300 random cases with 0–3
  unseen rows: |value − HiGHS LP| ≤ 4.4e-15): the dual value is max over λ ≥ 0 of −λε + Σ_s p_s min_t (f_t + λ C[s,t]),
  f = log.(Xt * a); for each s, walk the lower envelope of the lines f_t + λ·C[s,t]; then scan the breakpoints until the
  transport budget binds:
  ```julia
  f = log.(Xt * a); S, T = size(C)
  cur = zeros(Int, S); events = Tuple{Float64,Int,Int,Int}[]       # (λ, s, old t, new t)
  for s in 1:S
      fmin = minimum(f); cand = findall(t -> f[t] == fmin, 1:T)
      ℓ = cand[argmin(C[s, cand])]; cur[s] = ℓ; λc = 0.0
      while C[s, ℓ] > 0
          best, bt = Inf, 0
          for t in 1:T
              C[s, t] < C[s, ℓ] || continue
              λt = max((f[t] - f[ℓ]) / (C[s, ℓ] - C[s, t]), λc)
              if λt < best || (λt == best && C[s, t] < C[s, bt]); best, bt = λt, t; end
          end
          push!(events, (best, s, ℓ, bt)); ℓ = bt; λc = best
      end
  end
  budget = sum(p[s] * C[s, cur[s]] for s in 1:S)
  if budget ≤ ε                              # λ* = 0: the budget reaches every scenario's worst outcome
      Q = zeros(T); for s in 1:S; Q[cur[s]] += p[s]; end
      return (sum(Q .* f), 0.0, Q)
  end
  sort!(events, by = e -> e[1])
  asg = copy(cur); i = 1
  while i ≤ length(events)
      λb = events[i][1]; j = i
      while j ≤ length(events) && events[j][1] == λb; j += 1; end
      newbudget = budget + sum(p[events[k][2]] * (C[events[k][2], events[k][4]] - C[events[k][2], events[k][3]]) for k in i:j-1)
      if newbudget ≤ ε
          θ = (ε - newbudget) / (budget - newbudget)   # share of the switching mass left on the old outcome: budget binds exactly
          Q = zeros(T); sw = Set(events[k][2] for k in i:j-1)
          for s in 1:S; s in sw || (Q[asg[s]] += p[s]); end
          for k in i:j-1; s, o, n = events[k][2], events[k][3], events[k][4]; Q[o] += θ * p[s]; Q[n] += (1 - θ) * p[s]; end
          return (sum(Q .* f), λb, Q)
      end
      for k in i:j-1; asg[events[k][2]] = events[k][4]; end
      budget = newbudget; i = j
  end
  ```

  ball it can move mass onto `unseen` scenarios the model gives zero probability.

function kl_cash_radius(m::ScenarioModel; maxit = 2000, tol = 1e-13)
    m.has_cash || throw(ArgumentError("kl_cash_radius needs a model with cash"))
    pos = findall(>(0.0), m.π); p = m.π[pos]
    Y = m.X[pos, 2:end] .- 1.0; n = size(Y, 2); μ = zeros(n)
    φ(μ) = (z = -(Y * μ) .+ log.(p); mz = maximum(z); -(mz + log(sum(exp.(z .- mz)))))   # concave dual; = D_min at the optimum
    Qof(μ) = (lw = log.(p) .- Y * μ; lw .-= maximum(lw); Q = exp.(lw); Q ./ sum(Q))   # Q ∝ π·exp(−Σ μ_j (X_j − 1))
    kkt(μ, g) = max(maximum(abs, g[μ .> 0]; init = 0.0), maximum(g[μ .== 0]; init = 0.0))
    φmax = log(1 / minimum(p))                     # D(Q‖π) ≤ log(1/π_min) for every Q
    for _ in 1:maxit                               # projected Newton (Bertsekas) on max φ(μ), μ ≥ 0
        Q = Qof(μ); g = Y' * Q; k0 = kkt(μ, g)
        k0 ≤ tol && break
        F = findall(j -> μ[j] > 0 || g[j] > 0, 1:n)
        YF = Y[:, F]; c = YF' * Q
        H = -(YF' * (YF .* Q) .- c * c')           # = −Cov_Q(Y_F)
        d = zeros(n); d[F] = (H - (1e-12 + norm(g[F])) * I) \ (-g[F])   # Levenberg: ascent even when H is singular
        t = 1.0; φ0 = φ(μ)
        while t > 1e-16
            μn = max.(μ .+ t .* d, 0.0)
            if k0 > 1e-6
                φ(μn) ≥ φ0 + 1e-4 * dot(g, μn .- μ) && break      # far from the optimum: Armijo on φ
            else
                kkt(μn, Y' * Qof(μn)) < k0 && break               # near it φ is flat to rounding: accept a smaller KKT residual
            end
            t /= 2
        end
        μn = max.(μ .+ t .* d, 0.0); μn == μ && break; μ = μn
        # exact proofs that no no-edge world exists: the dual exceeds every possible D, or μ is a bet mix beating cash everywhere
        (φ(μ) > φmax + 1e-9 || minimum(Y * μ) > 0) && return (Inf, Qof(μ))
    end
    all(==(0.0), μ) && return (0.0, p)             # no bet has an edge under π
    Q = Qof(μ)
    minimum(Y * μ) > 0 && return (Inf, Q)
    cert = maximum(Y' * Q)
    cert > 1e-12 && throw(CertificationError(cert, 1e-12, :kl, "cash radius undecided (nearly degenerate model)"))
    return (max(kl(Q, p), 0.0), Q)
end
